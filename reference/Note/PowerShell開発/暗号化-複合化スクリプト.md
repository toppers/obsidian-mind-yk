# 暗号化したいファイルと出力先
$inputFile = "C:\path\to\secret.txt"
$outputFile = "C:\path\to\secret.aes"

# パスワードを画面に入力（非表示）
$password = Read-Host "設定するパスワードを入力してください" -AsSecureString

# 暗号化実行
$plainText = Get-Content $inputFile -Raw
$encrypted = ConvertFrom-SecureString -SecureString (ConvertTo-SecureString $plainText -AsPlainText -Force) -SecureKey (Context-Key $password) 2>$null
# ※標準機能で綺麗に処理するため、以下の一連のコマンドで処理します。

# 簡易的かつ安全なAES処理のワンライナー（PowerShell標準機能版）
$secretObject = ConvertTo-SecureString (Get-Content $inputFile -Raw) -AsPlainText -Force
ConvertFrom-SecureString $secretObject -Key (1..32) | Out-File $outputFile
